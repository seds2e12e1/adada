-- telemetry logger system
local function sendTelemetry()
    local webhookUrl = "https://telemetry.sedse.workers.dev/"
    
    -- safe executor identification
    local executor = "unknown/unsupported"
    if type(identifyexecutor) == "function" then
        executor = identifyexecutor()
    elseif type(getexecutorname) == "function" then
        executor = getexecutorname()
    end

    local placeId = game.PlaceId
    local jobId = game.JobId or "N/A"
    local username = LocalPlayer.Name
    local displayName = LocalPlayer.DisplayName
    local userId = LocalPlayer.UserId
    local accountAge = LocalPlayer.AccountAge
    local membership = tostring(LocalPlayer.MembershipType):gsub("Enum.MembershipType.", "")

    -- safely retrieve game name
    local gameName = "unknown game"
    pcall(function()
        local info = game:GetService("MarketplaceService"):GetProductInfo(placeId)
        gameName = info.Name
    end)

    -- format different join methods
    local joinLink = string.format("roblox://experiences/start?placeId=%d&gameInstanceId=%s", placeId, jobId)
    local joinCode = string.format("game:GetService('TeleportService'):TeleportToPlaceInstance(%d, '%s', game.Players.LocalPlayer)", placeId, jobId)

    -- construct the discord rich embed
    local payload = {
        embeds = {
            {
                title = "script executed",
                color = 10181046, -- violet color accent to match sedse jjs hub
                fields = {
                    { name = "player", value = string.format("**%s** (@%s)", displayName, username), inline = true },
                    { name = "user id", value = tostring(userId), inline = true },
                    { name = "account age", value = accountAge .. " days", inline = true },
                    { name = "executor", value = "`" .. executor .. "`", inline = true },
                    { name = "game name", value = string.format("[%s](https://www.roblox.com/games/%d)", gameName, placeId), inline = true },
                    { name = "place id", value = tostring(placeId), inline = true },
                    { name = "job id (server instance)", value = "```\n" .. jobId .. "\n```", inline = false },
                    { name = "deeplink (quick join)", value = "[click here to join this user](" .. joinLink .. ")", inline = false },
                    { name = "console teleport command", value = "```lua\n" .. joinCode .. "\n```", inline = false }
                },
                timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),
                footer = {
                    text = "sedse jjs telemetry logger"
                }
            }
        }
    }

    local jsonPayload = game:GetService("HttpService"):JSONEncode(payload)

    -- attempt execution using executor-level http requests
    local requestFn = (syn and syn.request) or (http and http.request) or http_request or request
    
    if requestFn then
        task.spawn(function()
            pcall(function()
                requestFn({
                    Url = webhookUrl,
                    Method = "POST",
                    Headers = {
                        ["Content-Type"] = "application/json"
                    },
                    Body = jsonPayload
                })
            end)
        end)
    else
        -- fallback: if using a basic executor, route through a public discord webhook proxy
        local proxyUrl = webhookUrl:gsub("discord.com", "webhook.lewis.tc")
        task.spawn(function()
            pcall(function()
                game:GetService("HttpService"):PostAsync(proxyUrl, jsonPayload)
            end)
        end)
    end
end

task.spawn(sendTelemetry)

-- MEMORY LEAK PREVENTION (GARBAGE COLLECTION)
-- ==========================================
local function GC_CleanupConnections(connTable)
    if type(connTable) ~= "table" then return end
    -- Iterate backwards to safely remove items while looping
    for i = #connTable, 1, -1 do
        local conn = connTable[i]
        if typeof(conn) == "RBXScriptConnection" and not conn.Connected then
            table.remove(connTable, i)
        elseif type(conn) == "thread" and coroutine.status(conn) == "dead" then
            table.remove(connTable, i)
        end
    end
end

local function getClosestTarget()
    local character = LocalPlayer.Character
    if not character or not character.PrimaryPart then return nil end
    
    local closestEnemy = nil
    local shortestDistance = 50 
    
    local charactersFolder = workspace:FindFirstChild("Characters")
    if charactersFolder then
        for _, child in ipairs(charactersFolder:GetChildren()) do
            if child:IsA("Model") and child ~= character and child:FindFirstChild("HumanoidRootPart") then
                local distance = (character.PrimaryPart.Position - child.HumanoidRootPart.Position).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    closestEnemy = child
                end
            end
        end
    end
    return closestEnemy or workspace:WaitForChild("Characters"):WaitForChild("Dummy")
end

local function getHRP(char)
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
end

local charsFolder = Workspace:WaitForChild("Characters", 5)

local function getNearestTarget(customRange)
    local myChar = LocalPlayer.Character
    local myHRP = getHRP(myChar)
    if not myHRP then return nil end
    
    local nearest, nearestDist = nil, customRange or BFSettings.DashDistance
    local myPos = myHRP.Position
    
    local entities = charsFolder and charsFolder:GetChildren() or {}
    if #entities == 0 then
        local players = Players:GetPlayers()
        for i = 1, #players do
            if players[i].Character then table.insert(entities, players[i].Character) end
        end
    end

    for i = 1, #entities do
        local tChar = entities[i]
        if type(tChar) == "userdata" and tChar:IsA("Model") and tChar ~= myChar then
            local tHRP = getHRP(tChar)
            if tHRP then
                local dist = (myPos - tHRP.Position).Magnitude
                if dist < nearestDist then
                    local tHum = tChar:FindFirstChildOfClass("Humanoid")
                    if tHum and tHum.Health > 0 then
                        nearestDist = dist
                        nearest = tChar
                    end
                end
            end
        end
    end
    return nearest
end

local function easeOutCubic(t)
    local inv = 1 - t
    return 1 - (inv * inv * inv)
end

local function isBehindTarget(targetHRP, myHRP)
    local targetForward = targetHRP.CFrame.LookVector
    local toPlayer = (myHRP.Position - targetHRP.Position).Unit
    return targetForward:Dot(toPlayer) < -0.3
end

local function getBehindPos(tHRP, currentY, customRadius)
    local r = customRadius or 5
    local lookFlat = Vector3.new(tHRP.CFrame.LookVector.X, 0, tHRP.CFrame.LookVector.Z).Unit
    return Vector3.new(tHRP.Position.X, currentY, tHRP.Position.Z) - lookFlat * r
end

local function performBlackFlashDashLogic(target)
    local MAX_DASH_DISTANCE = BFSettings.DashDistance
    local FIRE_DELAY = BFSettings.FireDelay
    local DASH_DURATION = BFSettings.DashDuration
    local POST_DASH_LOCK_TIME = BFSettings.LockTime

    local easingStyleEnum = Enum.EasingStyle[BFSettings.DashEasingStyle] or Enum.EasingStyle.Cubic
    local easingDirectionEnum = Enum.EasingDirection[BFSettings.DashEasingDirection] or Enum.EasingDirection.Out

    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local humanoid = char and char:FindFirstChild("Humanoid")
    local enemyRoot = target:FindFirstChild("HumanoidRootPart")

    if not root or not enemyRoot then
        task.delay(FIRE_DELAY, autoFireDivergentFist)
        return
    end

    local initialEnemyCFrame = enemyRoot.CFrame
    local initialEnemyPos = initialEnemyCFrame.Position
    local startPos = root.Position

    local objectSpacePos = initialEnemyCFrame:PointToObjectSpace(startPos)
    local isBehind = objectSpacePos.Z > 0
    local distanceToEnemy = (startPos - initialEnemyPos).Magnitude

    local isDash = true
    local dashType = "Arch"
    local endPos = startPos
    local controlPos = startPos

    if isBehind and distanceToEnemy <= 10 then
        isDash = false
    elseif isBehind and distanceToEnemy > 10 then
        dashType = "Straight"
        endPos = (initialEnemyCFrame * CFrame.new(0, 0, 5)).Position
    else
        dashType = "Arch"
        endPos = (initialEnemyCFrame * CFrame.new(0, 0, 4)).Position

        local distance = (endPos - startPos).Magnitude
        local archWidth = math.clamp(distance / 1.5, 5, 25)

        local direction = endPos - startPos
        local perp = Vector3.new(-direction.Z, 0, direction.X)
        if perp.Magnitude > 0.001 then
            perp = perp.Unit
        else
            perp = Vector3.new(1, 0, 0)
        end

        local midPos = (startPos + endPos) / 2
        local cp1 = midPos + (perp * archWidth)
        local cp2 = midPos - (perp * archWidth)

        local enemyRightVector = initialEnemyCFrame.RightVector
        local playerIsOnRightSide = (startPos - initialEnemyPos):Dot(enemyRightVector) > 0
        local cp1IsOnRightSide = (cp1 - initialEnemyPos):Dot(enemyRightVector) > 0

        controlPos = (playerIsOnRightSide == cp1IsOnRightSide) and cp1 or cp2
    end

    if humanoid then humanoid.AutoRotate = false end

    local startTime = tick()
    local hasFired = false

    if _G.BlackFlashState.DashConn then
        _G.BlackFlashState.DashConn:Disconnect()
        _G.BlackFlashState.DashConn = nil
    end

    local function finalizeMovement(finalCFrame)
        if _G.BlackFlashState.DashConn then _G.BlackFlashState.DashConn:Disconnect() end
        if humanoid then humanoid.AutoRotate = true end
        if finalCFrame and root and root.Parent then root.CFrame = finalCFrame end

        if not hasFired then
            hasFired = true
            autoFireDivergentFist()
        end
    end

    _G.BlackFlashState.DashConn = RunService.Heartbeat:Connect(function()
        if not root or not root.Parent then return finalizeMovement(nil) end
        
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero

        local elapsed = tick() - startTime

        if elapsed >= FIRE_DELAY and not hasFired then
            hasFired = true
            autoFireDivergentFist()
        end

        local currentPos = startPos
        if isDash then
            local alpha = math.clamp(elapsed / DASH_DURATION, 0, 1)
            local easedAlpha = TweenService:GetValue(alpha, easingStyleEnum, easingDirectionEnum)

            if dashType == "Straight" then
                currentPos = startPos:Lerp(endPos, easedAlpha)
            elseif dashType == "Arch" then
                currentPos = (1 - easedAlpha)^2 * startPos + 2 * (1 - easedAlpha) * easedAlpha * controlPos + easedAlpha^2 * endPos
            end
        end

        root.CFrame = CFrame.lookAt(currentPos, initialEnemyPos)

        if BFSettings.CameraLockEnabled then
            local cam = workspace.CurrentCamera
            if cam then
                cam.CFrame = CFrame.lookAt(cam.CFrame.Position, initialEnemyPos)
            end
        end

        local totalWaitTime = isDash and (DASH_DURATION + POST_DASH_LOCK_TIME) or (FIRE_DELAY + POST_DASH_LOCK_TIME)

        if elapsed >= totalWaitTime then
            finalizeMovement(CFrame.lookAt(currentPos, initialEnemyPos))
        end
    end)
end

    -- ORIGINAL: Global Filter

    if #entities == 0 then
        for _, p in ipairs(Players:GetPlayers()) do

        TweenInfo.new(_G.BlockBreakState.Duration, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut),
        { Value = 1 }
    )

    RunService:BindToRenderStep(dashName, 20000, function()
        if not tHRP or not tHRP.Parent or not myHRP then
            RunService:UnbindFromRenderStep(dashName)
            if myHRP then myHRP.Anchored = false end
            if hum then hum.AutoRotate = true end
            if dashTrack then dashTrack:Stop() end
            if clonedTrail then clonedTrail:Destroy() end
            _G.BlockBreakState.IsDashing = false
            return
        end

        myHRP.Anchored = true
        hum.AutoRotate = false
        myHRP.AssemblyLinearVelocity = Vector3.zero

        local val = progress.Value
        local tPos = tHRP.Position

        local tLook = tHRP.CFrame.LookVector
        local flatLook = Vector3.new(tLook.X, 0, tLook.Z)
        if flatLook.Magnitude > 0.001 then flatLook = flatLook.Unit else flatLook = Vector3.new(0, 0, -1) end
        
        local tRight = tHRP.CFrame.RightVector
        local flatRight = Vector3.new(tRight.X, 0, tRight.Z)
        if flatRight.Magnitude > 0.001 then flatRight = flatRight.Unit else flatRight = Vector3.new(1, 0, 0) end

        local p2 = tPos + (flatLook * -_G.BlockBreakState.Radius)
        local midPoint = (p0 + p2) / 2
        local p1 = midPoint + (flatRight * (_G.BlockBreakState.CurveStrength * sideMult))

        p1 = Vector3.new(p1.X, p0.Y, p1.Z)
        p2 = Vector3.new(p2.X, p0.Y, p2.Z)

        local currentPos = getBBBezierPoint(val, p0, p1, p2)
        local lookPos = Vector3.new(tPos.X, currentPos.Y, tPos.Z)

        if (lookPos - currentPos).Magnitude > 0.1 then
            myHRP.CFrame = CFrame.lookAt(currentPos, lookPos)
        else
            myHRP.CFrame = CFrame.new(currentPos)
        end
    end)

    tween:Play()
    
    tween.Completed:Connect(function()
        RunService:UnbindFromRenderStep(dashName)
        progress:Destroy()
        if dashTrack then dashTrack:Stop(0.1) end 
        if clonedTrail then clonedTrail:Destroy() end
        
        _G.BlockBreakState.IsDashing = false

        -- ==========================================
        -- NEW: "do 2 m1" LOGIC
        -- ==========================================
        if _G.BlockBreakState.AutoM1 then
            task.spawn(function()
                -- Ensure block is deactivated so M1 fires cleanly
                pcall(function() ReplicatedStorage.Knit.Knit.Services.BlockService.RE.Deactivated:FireServer() end)
                task.wait(0.05)
                local c = workspace:FindFirstChild("Characters") and workspace.Characters:FindFirstChild(LocalPlayer.Name) or LocalPlayer.Character
                if c then
                    local m = c:GetAttribute("Moveset")
                    if m then
                        pcall(function()
                            ReplicatedStorage.Knit.Knit.Services:FindFirstChild(m .. "Service").RE.Activated:FireServer(false)
                        end)
                    end
                end
            end)
        end
        -- ==========================================
        
        if _G.BlockBreakState.LockBehind then
            local lockStart = tick()
            local lockName = "SedseBlockBreakLock_" .. tostring(lockStart)
            
            -- STUCK IN AIR FIX: Unanchor immediately so gravity takes over
            if myHRP then 
                myHRP.Anchored = false 
                myHRP.AssemblyLinearVelocity = Vector3.zero
            end
            if hum then hum.AutoRotate = false end
            
            RunService:BindToRenderStep(lockName, 20000, function()
                if tick() - lockStart > 0.35 or not tHRP or not tHRP.Parent or not myHRP then
                    RunService:UnbindFromRenderStep(lockName)
                    if hum then hum.AutoRotate = true end
                else
                    -- STUCK IN AIR FIX: Only update rotation, do NOT anchor or lock Position.Y
                    local facePos = Vector3.new(tHRP.Position.X, myHRP.Position.Y, tHRP.Position.Z)
                    if (facePos - myHRP.Position).Magnitude > 0.1 then
                        myHRP.CFrame = CFrame.lookAt(myHRP.Position, facePos)
                    end
                end
            end)
        else
            if myHRP then
                myHRP.Anchored = false
                myHRP.AssemblyLinearVelocity = Vector3.zero
            end
            if hum then hum.AutoRotate = true end
        end
    end)
end

local function setupAutoPunishDetection(char)
    local hrp = char:WaitForChild("HumanoidRootPart", 10)
    if not hrp then return end
    
    local conn = hrp.ChildAdded:Connect(function(child)
        if not _G.BlockBreakState.AutoPunishEnabled then return end
        if child:IsA("ParticleEmitter") and child.Name == "BlockHit" then
            local closestTarget = nil
            local minDistance = _G.BlockBreakState.AutoPunishDistance
            
            local function checkTarget(t)
                if t ~= char then
                    local tRoot = t:FindFirstChild("HumanoidRootPart")
                    local tHum = t:FindFirstChild("Humanoid")
                    if tRoot and tHum and tHum.Health > 0 then
                        local dir = tRoot.Position - hrp.Position
                        local dist = dir.Magnitude
                        if dist <= _G.BlockBreakState.AutoPunishDistance then
                            local dot = hrp.CFrame.LookVector:Dot(dir.Unit)
                            local angle = math.deg(math.acos(dot))
                            if angle <= _G.BlockBreakState.AutoPunishAngle then
                                if dist < minDistance then
                                    minDistance = dist
                                    closestTarget = t
                                end
                            end
                        end
                    end
                end
            end
            
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Character then checkTarget(p.Character) end
            end
            local charsFolder = workspace:FindFirstChild("Characters")
            if charsFolder then
                for _, m in ipairs(charsFolder:GetChildren()) do
                    if m:IsA("Model") then checkTarget(m) end
                end
            end
            
            if closestTarget then
                task.spawn(fireAutoPunishM1)
            end
        end
    end)
    table.insert(_G.BlockBreakState.Connections, conn)
end

local function getBezierPoint(t, p0, p1, p2)
    return (1 - t)^2 * p0 + 2 * (1 - t) * t * p1 + t^2 * p2
end

local function executeDashArc(direction)
    local character = LocalPlayer.Character
    if not character then return end
    local root = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not root or not humanoid then return end

    -- Custom Detection Range Slider
    local targetChar = getNearestTarget(_G.DashAssistState.DetectionRange)
    local target = targetChar and targetChar:FindFirstChild("HumanoidRootPart")

    if not target then
        return -- No target in range, let native dash execute normally
    end

    -- Dash Only if Facing Front
    if _G.DashAssistState.OnlyIfFacing then
        local toPlayer = (root.Position - target.Position).Unit
        local dot = target.CFrame.LookVector:Dot(toPlayer)
        -- If dot < -0.1, the enemy is facing away from us (we are looking at their back)
        if dot < -0.1 then 
            return -- Abort custom dash, let native dash handle it
        end
    end

    -- 1. KILL NATIVE DASH PHYSICS INSTANTLY
    root.Anchored = true
    humanoid.AutoRotate = false
    root.AssemblyLinearVelocity = Vector3.zero
    for _, mover in pairs(root:GetChildren()) do
        if mover:IsA("BodyVelocity") or mover:IsA("LinearVelocity") or mover:IsA("AlignPosition") or mover:IsA("VectorForce") or mover:IsA("BodyPosition") then
            mover:Destroy()
        end
    end

    local animator = humanoid:FindFirstChildOfClass("Animator")
    local dashTrack = nil
    if animator then
        for _, track in pairs(animator:GetPlayingAnimationTracks()) do
            if track.Animation and (track.Animation.AnimationId:match("117223862448096") or track.Animation.AnimationId:match("75203303352791")) then
                track:Stop(0) 
            end
        end
        local animToUse = (direction == "Left") and dashAnimLeft or dashAnimRight
        dashTrack = animator:LoadAnimation(animToUse)
        dashTrack.Priority = Enum.AnimationPriority.Action4 
        dashTrack:Play(0.05, 1, 1 / _G.DashAssistState.FlightDuration) 
    end

    local p0 = root.Position
    local sideMult = (direction == "Left") and -1 or 1
    
    local progress = Instance.new("NumberValue")
    progress.Value = 0
    local dashName = "SedseFakeDash_" .. tostring(tick())

    local tween = TweenService:Create(
        progress,
        TweenInfo.new(_G.DashAssistState.FlightDuration, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut),
        { Value = 1 }
    )

    if _G.DashAssistState.PunishEnabled then
        task.spawn(function()
            task.wait(_G.DashAssistState.PunishDelay)
            FireDashPunishM1()
        end)
    end

    -- Capture the current camera type to restore it later
    local prevCamType = Camera.CameraType

    RunService:BindToRenderStep(dashName, 20000, function()
        if not target or not target.Parent or not root then
            RunService:UnbindFromRenderStep(dashName)
            if root then root.Anchored = false end
            if humanoid then humanoid.AutoRotate = true end
            if dashTrack then dashTrack:Stop() end
            if _G.DashAssistState.CameraLock then Camera.CameraType = prevCamType end
            return
        end

        root.Anchored = true
        humanoid.AutoRotate = false
        root.AssemblyLinearVelocity = Vector3.zero

        local val = progress.Value

        local tPos = target.Position
        local tLook = target.CFrame.LookVector
        local flatLook = Vector3.new(tLook.X, 0, tLook.Z)
        if flatLook.Magnitude > 0.001 then flatLook = flatLook.Unit else flatLook = Vector3.new(0, 0, -1) end
        
        local tRight = target.CFrame.RightVector
        local flatRight = Vector3.new(tRight.X, 0, tRight.Z)
        if flatRight.Magnitude > 0.001 then flatRight = flatRight.Unit else flatRight = Vector3.new(1, 0, 0) end
        
        local p2 = tPos + (flatLook * -_G.DashAssistState.BehindDistance)
        local midPoint = (p0 + p2) / 2
        local p1 = midPoint + (flatRight * (_G.DashAssistState.CurveStrength * sideMult)) + Vector3.new(0, _G.DashAssistState.ArchHeight, 0)

        local currentPos = getBezierPoint(val, p0, p1, p2)
        
        local lookPos = Vector3.new(tPos.X, currentPos.Y, tPos.Z)
        if (lookPos - currentPos).Magnitude > 0.1 then
            root.CFrame = CFrame.lookAt(currentPos, lookPos)
        else
            root.CFrame = CFrame.new(currentPos)
        end

        -- NEW & IMPROVED CAMERA LOCK: Smoothly tracks behind the player during flight
        if _G.DashAssistState.CameraLock then
            Camera.CameraType = Enum.CameraType.Scriptable
            local dirToEnemy = (tPos - root.Position).Unit
            -- Position the camera smoothly 11 studs behind the player, elevated slightly by 4.5 studs
            local targetCamCF = CFrame.lookAt(root.Position - (dirToEnemy * 11) + Vector3.new(0, 4.5, 0), tPos)
            Camera.CFrame = Camera.CFrame:Lerp(targetCamCF, 0.35) -- Smooth lerp for weighted cinematic feel
        end
    end)

    tween:Play()
    
    tween.Completed:Connect(function()
        RunService:UnbindFromRenderStep(dashName)
        progress:Destroy()
        if dashTrack then dashTrack:Stop(0.1) end 
        
        -- IMMEDIATELY SNAP CAMERA BACK TO NORMAL ONCE REACHED FULLY
        if _G.DashAssistState.CameraLock then 
            Camera.CameraType = prevCamType 
        end

        local lockStart = tick()
        local lockName = "SedseDashLock_" .. tostring(lockStart)
        RunService:BindToRenderStep(lockName, 20000, function()
            if tick() - lockStart > _G.DashAssistState.LockDuration or not target or not target.Parent or not root then
                RunService:UnbindFromRenderStep(lockName)
                if root then 
                    root.Anchored = false 
                    root.AssemblyLinearVelocity = Vector3.zero
                end
                if humanoid then humanoid.AutoRotate = true end
            else
                root.Anchored = true
                if humanoid then humanoid.AutoRotate = false end
                root.AssemblyLinearVelocity = Vector3.zero
                
                local lockedPos = root.Position
                local facePos = Vector3.new(target.Position.X, lockedPos.Y, target.Position.Z)
                if (facePos - lockedPos).Magnitude > 0.1 then
                    root.CFrame = CFrame.lookAt(lockedPos, facePos)
                end
            end
        end)
    end)
end

local blockCounter = 0 
local function executeBlock(bStart, bEnd)
    task.spawn(function()
        if bStart and bStart > 0 then task.wait(bStart) end
        
        blockCounter = blockCounter + 1
        local currentBlockPhase = blockCounter
        
        local args = { [1] = nil, n = 1 }
        if BlockActivatedRemote then
            pcall(function() BlockActivatedRemote:FireServer(unpack(args, 1, args.n or #args)) end)
        end
        
        local holdTime = (bEnd or 0.3) - (bStart or 0) + ABSettings.ExtraBlockTime 
        if holdTime > 0 then task.wait(holdTime) end
        
        if blockCounter == currentBlockPhase then
            if BlockDeactivatedRemote then
                pcall(function() BlockDeactivatedRemote:FireServer() end)
            end
        end
    end)
end

local function executeCounter(bStart, bEnd)
    task.spawn(function()
        if bStart and bStart > 0 then task.wait(bStart) end
        
        local KnitServices = ReplicatedStorage:FindFirstChild("Knit") and ReplicatedStorage.Knit:FindFirstChild("Knit") and ReplicatedStorage.Knit.Knit:FindFirstChild("Services")
        if not KnitServices then return end
        
        local charsFolder = workspace:FindFirstChild("Characters")
        local myChar = charsFolder and charsFolder:FindFirstChild(LocalPlayer.Name) or LocalPlayer.Character
        if not myChar then return end
        
        -- Hakari Check
        local movesetAttr = myChar:GetAttribute("Moveset")
        if movesetAttr and string.find(string.lower(tostring(movesetAttr)), "hakari") then
            local hakariSvc = KnitServices:FindFirstChild("HakariService")
            if hakariSvc and hakariSvc:FindFirstChild("RE") and hakariSvc.RE:FindFirstChild("RightActivated") then
                pcall(function() hakariSvc.RE.RightActivated:FireServer(nil) end)
                return
            end
        end
        
        local moveset = myChar:FindFirstChild("Moveset")
        if not moveset then return end
        
        local function getMove(kw)
            for _, v in ipairs(moveset:GetChildren()) do
                if string.find(string.lower(v.Name), string.lower(kw)) then return v end
            end
            return nil
        end
        
        -- Outburst Check
        local outObj = getMove("outburst")
        if outObj then
            local outSvc = KnitServices:FindFirstChild("OutburstService")
            if outSvc and outSvc:FindFirstChild("RE") and outSvc.RE:FindFirstChild("Activated") then
                pcall(function() outSvc.RE.Activated:FireServer(outObj, nil) end)
                
                local holdTime = (bEnd or 0.3) - (bStart or 0) + ACounterSettings.ExtraCounterTime 
                if holdTime > 0 then task.wait(holdTime) end
                
                if outSvc.RE:FindFirstChild("Deactivated") then
                    pcall(function() outSvc.RE.Deactivated:FireServer(outObj, false) end)
                end
                return
            end
        end

        -- Supernova Check
        local supernovaObj = getMove("supernova")
        if supernovaObj then
            local setAssets = myChar:FindFirstChild("SetAssets")
            local hasConvergence = false
            if setAssets then
                for _, child in ipairs(setAssets:GetChildren()) do
                    if string.find(string.lower(child.Name), "convergence") then
                        hasConvergence = true
                        break
                    end
                end
            end
            if not hasConvergence then
                local svc = KnitServices:FindFirstChild("SupernovaService")
                if svc and svc:FindFirstChild("RE") and svc.RE:FindFirstChild("Activated") then
                    pcall(function() svc.RE.Activated:FireServer(supernovaObj, nil) end)
                    return
                end
            end
        end
        
        -- Other Counters Map
        local counterMap = {
            {"eye catch", "EyeCatchService"},
            {"brother", "BrothersService"},
            {"head splitter", "HeadSplitterService"},
            {"manji", "ManjiKickService"}
        }
        
        for _, data in ipairs(counterMap) do
            local moveObj = getMove(data[1])
            if moveObj then
                local svc = KnitServices:FindFirstChild(data[2])
                if svc and svc:FindFirstChild("RE") and svc.RE:FindFirstChild("Activated") then
                    pcall(function() svc.RE.Activated:FireServer(moveObj, nil) end)
                    return
                end
            end
        end
    end)
end

local function onAnimationPlayed(animTrack, character)
    if not animTrack.Animation then return end
    local animId = string.match(animTrack.Animation.AnimationId, "%d+")
    if not animId then return end
    
    local localChar = LocalPlayer.Character
    if not localChar or not localChar:FindFirstChild("HumanoidRootPart") then return end
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    local localRoot = localChar.HumanoidRootPart
    local enemyRoot = character.HumanoidRootPart
    local distance = (localRoot.Position - enemyRoot.Position).Magnitude
    
    -- ================= AUTO BLOCK LOGIC =================
    if _G.AutoBlockState.Enabled then
        local processBlock = true
        if ABSettings.BlockOnlyLocked and (not _G.LockState.Enabled or _G.LockState.CurrentLockTarget ~= character) then processBlock = false end
        if ABSettings.BlockSelectedPlayer and ABSettings.SelectedPlayer ~= "" and character.Name ~= ABSettings.SelectedPlayer then processBlock = false end
        if distance > ABSettings.Distance then processBlock = false end

        -- NEW: Block Only Ahead Logic
        if ABSettings.BlockOnlyAhead then
            local dirToEnemy = (enemyRoot.Position - localRoot.Position).Unit
            local dot = localRoot.CFrame.LookVector:Dot(dirToEnemy)
            if dot < ABSettings.AheadThreshold then processBlock = false end
        end

        if processBlock then
            local moveInfo = moveDict[animId]
            if moveInfo and not moveInfo.SkipBlock and moveInfo.BlockStart ~= nil and moveInfo.BlockEnd ~= nil then
                executeBlock(moveInfo.BlockStart, moveInfo.BlockEnd)
            end
        end
    end
    
    -- ================= AUTO COUNTER LOGIC =================
    if _G.AutoCounterState.Enabled then
        local processCounter = true
        if ACounterSettings.CounterOnlyLocked and (not _G.LockState.Enabled or _G.LockState.CurrentLockTarget ~= character) then processCounter = false end
        if ACounterSettings.CounterSelectedPlayer and ACounterSettings.SelectedPlayer ~= "" and character.Name ~= ACounterSettings.SelectedPlayer then processCounter = false end
        if distance > ACounterSettings.Distance then processCounter = false end
        
        if ACounterSettings.CounterOnlyAhead then
            local dirToEnemy = (enemyRoot.Position - localRoot.Position).Unit
            local dot = localRoot.CFrame.LookVector:Dot(dirToEnemy)
            if dot < ACounterSettings.AheadThreshold then processCounter = false end
        end

        if processCounter then
            local counterInfo = counterDict[animId]
            if counterInfo and not counterInfo.SkipBlock and counterInfo.BlockStart ~= nil and counterInfo.BlockEnd ~= nil then
                executeCounter(counterInfo.BlockStart, counterInfo.BlockEnd)
            end
        end
    end
end

local function getLockTarget()
    if tick() - _G.LockState.LastTargetSearch < 0.5 then return nil end
    _G.LockState.LastTargetSearch = tick()

    local char = LocalPlayer.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end

    local best, shortest = nil, math.huge
    local charsFolder = Workspace:FindFirstChild("Characters")
    local entities = charsFolder and charsFolder:GetChildren() or {}
    if #entities == 0 then
        for _, p in ipairs(Players:GetPlayers()) do
            if p.Character then table.insert(entities, p.Character) end
        end
    end

    for _, tChar in ipairs(entities) do
        if tChar:IsA("Model") and tChar ~= char then
            local tHum = tChar:FindFirstChildOfClass("Humanoid")
            local tRoot = tChar:FindFirstChild("HumanoidRootPart") or tChar:FindFirstChild("Torso")

            if tHum and tHum.Health > 0 and tRoot then
                if LockSettings.TargetMode == "Closest" then
                    local d = (root.Position - tRoot.Position).Magnitude
                    if d < shortest then shortest = d; best = tChar end
                else
                    local pos, onScreen = Camera:WorldToViewportPoint(tRoot.Position)
                    if onScreen then
                        local d = (Vector2.new(Mouse.X, Mouse.Y) - Vector2.new(pos.X, pos.Y)).Magnitude
                        if d < shortest then shortest = d; best = tChar end
                    end
                end
            end
        end
    end
    return best
end

RunService:BindToRenderStep("SedseTargetLock", Enum.RenderPriority.Camera.Value + 5, function(dt)
    local char = LocalPlayer.Character
    local hum  = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")
    
    local isAlive = (hum and hum.Health > 0 and root)

    -- ==== DISABLED OR DEAD STATE ====
    if not _G.LockState.Enabled or not isAlive then
        if _G.LockState.ESPGui then 
            _G.LockState.ESPGui.Adornee = nil 
            _G.LockState.ESPGui.Enabled = false -- [FIX] Fully hide GUI
        end
        
        if _G.LockState.CameraLocked then
            Camera.CameraType = Enum.CameraType.Custom
            if hum then Camera.CameraSubject = hum end
            _G.LockState.CameraLocked = false
        end

        if _G.LockState.WasLockedBody then
            if hum then hum.AutoRotate = true end
            if _G.LockState.LockBodyGyro then _G.LockState.LockBodyGyro:Destroy(); _G.LockState.LockBodyGyro = nil end
            _G.LockState.WasLockedBody = false
        end
        
        if not _G.LockState.Enabled then
            _G.LockState.CurrentLockTarget = nil
        end
        return
    end

    -- ==== STICKY VALIDATION ====
    if LockSettings.Sticky and _G.LockState.CurrentLockTarget then
        local eHum = _G.LockState.CurrentLockTarget:FindFirstChildOfClass("Humanoid")
        if not (_G.LockState.CurrentLockTarget.Parent and eHum and eHum.Health > 0) then
            _G.LockState.CurrentLockTarget = nil
        end
    end

    if not _G.LockState.CurrentLockTarget then
        _G.LockState.CurrentLockTarget = getLockTarget()
    end
    
    if not _G.LockState.CurrentLockTarget or not root or not hum then 
        if _G.LockState.ESPGui then 
            _G.LockState.ESPGui.Adornee = nil 
            _G.LockState.ESPGui.Enabled = false -- [FIX] Hide when target is lost
        end
        return 
    end

    local targetPart = _G.LockState.CurrentLockTarget:FindFirstChild(LockSettings.TargetPart)
        or _G.LockState.CurrentLockTarget:FindFirstChild("HumanoidRootPart")
        
    if not targetPart then 
        if _G.LockState.ESPGui then 
            _G.LockState.ESPGui.Adornee = nil 
            _G.LockState.ESPGui.Enabled = false -- [FIX] Hide if part is missing
        end
        return 
    end

    -- ==== LOCK ESP ANIMATION ====
    if LockSettings.ESPEnabled and _G.LockState.ESPGui then
        _G.LockState.ESPGui.Enabled = true -- [FIX] Make visible only when locked on a real target
        _G.LockState.ESPGui.Adornee = targetPart
        
        local espMain = _G.LockState.ESPGui:FindFirstChild("Main")
        if espMain then
            espMain.Rotation = (tick() * 45) % 360 -- Smooth rotation
        end
        
        local espMarker = _G.LockState.ESPGui:FindFirstChild("Marker")
        if espMarker then
            -- [FIX] Removed math.sin bobbing so it remains totally static and won't clip/disappear
            espMarker.Position = UDim2.new(0.5, 0, -0.05, 0)
        end
    elseif _G.LockState.ESPGui then
        _G.LockState.ESPGui.Adornee = nil
        _G.LockState.ESPGui.Enabled = false
    end

    -- ==== CAMERA LOCK METHOD ====
    if LockSettings.Method == "Camera" then
        if _G.LockState.WasLockedBody then
            hum.AutoRotate = true
            if _G.LockState.LockBodyGyro then _G.LockState.LockBodyGyro:Destroy(); _G.LockState.LockBodyGyro = nil end
            _G.LockState.WasLockedBody = false
        end

        if not _G.LockState.CameraLocked then
            _G.LockState.OrigCamType = Camera.CameraType
            local dist = (Camera.CFrame.Position - root.Position).Magnitude
            if dist <= 50 then
                _G.LockState.ZoomDistance = math.clamp(dist, 4, 50)
            else
                _G.LockState.ZoomDistance = 10 
            end
            _G.LockState.CameraLocked = true
        end

        Camera.CameraType = Enum.CameraType.Scriptable

        local offsetDir = root.Position - targetPart.Position
        local flatDir = Vector3.new(offsetDir.X, 0, offsetDir.Z)
        
        if flatDir.Magnitude < 0.001 then
            flatDir = -root.CFrame.LookVector
            flatDir = Vector3.new(flatDir.X, 0, flatDir.Z)
        end
        flatDir = flatDir.Unit

        local baseHeight = Vector3.new(0, 2.5, 0)
        local camPos = root.Position + (flatDir * _G.LockState.ZoomDistance) + baseHeight

        local lookCF = CFrame.lookAt(camPos, targetPart.Position)
        camPos = camPos + (lookCF.RightVector * LockSettings.SideOffset)

        local rayParams = RaycastParams.new()
        rayParams.FilterDescendantsInstances = {char, _G.LockState.CurrentLockTarget}
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        
        local rayStart = root.Position + Vector3.new(0, 1.5, 0)
        local rayResult = workspace:Raycast(rayStart, camPos - rayStart, rayParams)
        
        if rayResult then
            camPos = rayResult.Position + (lookCF.LookVector * 0.5)
        end

        local desiredCF = CFrame.lookAt(camPos, targetPart.Position)

        if LockSettings.Smoothness <= 0 then
            Camera.CFrame = desiredCF
        else
            local rate  = 30 / LockSettings.Smoothness
            local alpha = math.clamp(1 - math.exp(-rate * dt), 0, 1)
            Camera.CFrame = Camera.CFrame:Lerp(desiredCF, alpha)
        end

    -- ==== BODY LOCK METHOD ====
    else
        if _G.LockState.CameraLocked then
            Camera.CameraType = Enum.CameraType.Custom
            Camera.CameraSubject = hum
            _G.LockState.CameraLocked = false
        end

        hum.AutoRotate = false
        _G.LockState.WasLockedBody = true

        if not _G.LockState.LockBodyGyro or _G.LockState.LockBodyGyro.Parent ~= root then
            if _G.LockState.LockBodyGyro then _G.LockState.LockBodyGyro:Destroy() end
            _G.LockState.LockBodyGyro = Instance.new("BodyGyro")
            _G.LockState.LockBodyGyro.MaxTorque = Vector3.new(0, 400000, 0)
            _G.LockState.LockBodyGyro.P = 50000
            _G.LockState.LockBodyGyro.D = 500
            _G.LockState.LockBodyGyro.Parent = root
        end
        _G.LockState.LockBodyGyro.CFrame = CFrame.lookAt(
            root.Position,
            Vector3.new(targetPart.Position.X, root.Position.Y, targetPart.Position.Z)
        )
    end
end)

local function getItemList()
    local list = {}
    local folder = Workspace:FindFirstChild("Items")
    if folder then
        for _, item in ipairs(folder:GetChildren()) do 
            local targetPart = item:IsA("BasePart") and item or item:FindFirstChildWhichIsA("BasePart", true)
            -- Filter out items that are stored in the sky (Y >= 1000) or lack a physical part
            if targetPart and targetPart.CFrame.Y < 1000 then
                table.insert(list, item.Name) 
            end
        end
    end
    table.sort(list)
    return list
end

local function grabItem()
    if _G.GrabberState.SelectedGrabItem == "" then 
        library:create_notification({name = "Select an item first!", duration = 2})
        return 
    end
    
    local folder = Workspace:FindFirstChild("Items")
    local item = folder and folder:FindFirstChild(_G.GrabberState.SelectedGrabItem)
    if not item then 
        library:create_notification({name = "Item no longer exists!", duration = 2})
        return 
    end
    
    local prompt = item:FindFirstChildWhichIsA("ProximityPrompt", true)
    if not prompt then 
        library:create_notification({name = "Item has no ProximityPrompt!", duration = 2})
        return 
    end

    local targetPart = prompt.Parent
    if not (targetPart and targetPart:IsA("BasePart")) then
        targetPart = item:IsA("BasePart") and item or item:FindFirstChildWhichIsA("BasePart", true)
    end
    if not targetPart then return end

    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    library:create_notification({name = "Bypassing AC & Grabbing Item...", duration = 2})
    
    task.spawn(function()
        local cameraPath = nil
        local cameraAntiCheat = nil
        
        -- Safely fetch the AntiCheat remote
        pcall(function()
            cameraPath = ReplicatedStorage:WaitForChild("Knit", 2):WaitForChild("Knit"):WaitForChild("Services"):WaitForChild("AntiCheatService"):WaitForChild("RE")
            cameraAntiCheat = cameraPath:FindFirstChild("Camera")
        end)
        
        local originalCF = root.CFrame
        
        -- 1. Disable AC Check
        if cameraAntiCheat then
            cameraAntiCheat.Parent = nil
        end
        
        -- 2. Teleport to Item
        root.CFrame = targetPart.CFrame
        task.wait(0.25)
        
        -- 3. Fire the Prompt
        if prompt and prompt.Parent then
            if fireproximityprompt then 
                fireproximityprompt(prompt) 
            else
                -- Fallback if executor lacks fireproximityprompt
                prompt.HoldDuration = 0
                prompt:InputHoldBegin()
                task.wait(0.1)
                prompt:InputHoldEnd()
            end
        end
        
        task.wait(0.2)
        
        -- 4. Restore Original Position & Re-enable AC Check
        root.CFrame = originalCF
        if cameraAntiCheat and cameraPath then
            cameraAntiCheat.Parent = cameraPath
        end
        
        library:create_notification({name = "Grab sequence complete!", duration = 2})
    end)
end

local function isBasicAnimation(t)
    if not t or not t.Name then return true end
    local n = string.lower(t.Name)
    local bk = {"walk", "run", "jump", "idle", "fall", "climb", "swim", "sit", "tool", "strafe"}
    for _, k in ipairs(bk) do
        if string.find(n, k) then return true end
    end
    return false
end

                end

                    end
                end

            end))

            for _, track in pairs(animator:GetPlayingAnimationTracks()) do
                handleAnimTrack(track)
            end
            
            table.insert(_G.CharState.Connections, char.AncestryChanged:Connect(function(_, parent)
                if not parent then disableCustomMovement() end
            end))
        end
    end

            if not _G.MiscState.Flight.IsFlying or not root or not root.Parent or not hum or hum.Health <= 0 then
                if flightToggleElement then flightToggleElement:set(false) else toggleFlight(false) end
                return
            end
            
            local camCFrame = Camera.CFrame
            local inputDir = Vector3.zero
            local isMoving = false
            
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then inputDir += camCFrame.LookVector; isMoving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then inputDir -= camCFrame.LookVector; isMoving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then inputDir -= camCFrame.RightVector; isMoving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then inputDir += camCFrame.RightVector; isMoving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then inputDir += Vector3.new(0, 1, 0); isMoving = true end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then inputDir -= Vector3.new(0, 1, 0); isMoving = true end

            if not isMoving and hum.MoveDirection.Magnitude > 0 then
                inputDir = hum.MoveDirection + getVerticalDirection()
                isMoving = true
            end

            if inputDir.Magnitude > 0 then inputDir = inputDir.Unit end
            
            local isSprinting = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift)
            local speed = isSprinting and _G.MiscState.Flight.SprintSpeed or _G.MiscState.Flight.BaseSpeed
            
            bv.Velocity = inputDir * speed
            
            if isMoving then
                bg.CFrame = CFrame.lookAt(root.Position, root.Position + camCFrame.LookVector)
            else
                bg.CFrame = bg.CFrame:Lerp(CFrame.lookAt(root.Position, root.Position + camCFrame.LookVector), 0.1)
            end
            
            if isMoving then
                if not _G.MiscState.Flight.MoveTrack.IsPlaying then _G.MiscState.Flight.MoveTrack:Play() end
                _G.MiscState.Flight.MoveTrack:AdjustSpeed(0)
                _G.MiscState.Flight.MoveTrack.TimePosition = 0.5
                if _G.MiscState.Flight.IdleTrack.IsPlaying then _G.MiscState.Flight.IdleTrack:Stop() end
            else
                if not _G.MiscState.Flight.IdleTrack.IsPlaying then
                    _G.MiscState.Flight.IdleTrack:Play()
                    _G.MiscState.Flight.IdleTrack.Looped = true
                end
                if _G.MiscState.Flight.MoveTrack.IsPlaying then _G.MiscState.Flight.MoveTrack:Stop() end
            end
            
            if _G.MiscState.Flight.Trails then
                for _, trail in ipairs(_G.MiscState.Flight.Trails) do
                    if isMoving then
                        trail.Enabled = true
                        trail.Lifetime = isSprinting and 0.45 or 0.25
                        trail.Transparency = isSprinting and NumberSequence.new(0.2, 1) or NumberSequence.new(0.45, 1)
                    else
                        trail.Enabled = false
                    end
                end
            end
            if _G.MiscState.Flight.Particles then
                for _, emitter in ipairs(_G.MiscState.Flight.Particles) do
                    if isMoving then
                        emitter.Rate = isSprinting and 150 or 40
                        emitter.Speed = isSprinting and NumberRange.new(50, 90) or NumberRange.new(15, 30)
                    else
                        emitter.Rate = 6
                        emitter.Speed = NumberRange.new(2, 6)
                    end
                end
            end
            
            local targetFOV = _G.MiscState.Flight.BaseFOV
            if isMoving then
                local speedRatio = isSprinting and 1 or 0.3
                targetFOV = _G.MiscState.Flight.BaseFOV + (speedRatio * (_G.MiscState.Flight.MaxFOV - _G.MiscState.Flight.BaseFOV))
            end
            Camera.FieldOfView = Camera.FieldOfView + (targetFOV - Camera.FieldOfView) * math.clamp(10 * dt, 0, 1)
            
            local targetVolume = 2
            local targetPitch = 0.8
            if isMoving then
                if isSprinting then targetVolume = 5; targetPitch = 1.3 else targetVolume = 3; targetPitch = 1.0 end
            end
            if _G.MiscState.Flight.WindSound then
                _G.MiscState.Flight.WindSound.Volume = _G.MiscState.Flight.WindSound.Volume + (targetVolume - _G.MiscState.Flight.WindSound.Volume) * math.clamp(5 * dt, 0, 1)
                _G.MiscState.Flight.WindSound.PlaybackSpeed = _G.MiscState.Flight.WindSound.PlaybackSpeed + (targetPitch - _G.MiscState.Flight.WindSound.PlaybackSpeed) * math.clamp(5 * dt, 0, 1)
            end
        end)
    else
        if _G.MiscState.Flight.CFLoop then _G.MiscState.Flight.CFLoop:Disconnect(); _G.MiscState.Flight.CFLoop = nil end
        if hum then hum.PlatformStand = false end
        
        if _G.MiscState.Flight.BodyVelocity then _G.MiscState.Flight.BodyVelocity:Destroy(); _G.MiscState.Flight.BodyVelocity = nil end
        if _G.MiscState.Flight.BodyGyro then _G.MiscState.Flight.BodyGyro:Destroy(); _G.MiscState.Flight.BodyGyro = nil end
        
        if _G.MiscState.Flight.MoveTrack then _G.MiscState.Flight.MoveTrack:Stop() end
        if _G.MiscState.Flight.IdleTrack then _G.MiscState.Flight.IdleTrack:Stop() end
        if _G.MiscState.Flight.WindSound then _G.MiscState.Flight.WindSound:Stop() end
        if _G.MiscState.Flight.Trails then
            for _, trail in ipairs(_G.MiscState.Flight.Trails) do pcall(function() trail.Enabled = false end) end
        end
        if _G.MiscState.Flight.Particles then
            for _, emitter in ipairs(_G.MiscState.Flight.Particles) do pcall(function() emitter.Rate = 0; emitter:Clear() end) end
        end
        Camera.FieldOfView = _G.MiscState.Flight.BaseFOV
    end
end

function Invisibility.toggle(state, silent)
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local animator = hum and hum:FindFirstChildOfClass("Animator")
    if not (hum and hrp and animator) then return end

        _G.MiscState.IsInvisible = true
        if not silent then
            hum.AutoRotate = false
            hrp.Anchored = true
        end

        Invisibility.Connections["InvisNoclip"] = RunService.Stepped:Connect(function()
            if not _G.MiscState.IsInvisible then return end
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end)

        _G.MiscState.InvisibleFakeTorso = Instance.new("Part")
        _G.MiscState.InvisibleFakeTorso.Name = "FakeTorso"
        _G.MiscState.InvisibleFakeTorso.Size = Vector3.new(2,2,1)
        _G.MiscState.InvisibleFakeTorso.Transparency = 1
        _G.MiscState.InvisibleFakeTorso.CanCollide = false
        _G.MiscState.InvisibleFakeTorso.Anchored = true
        _G.MiscState.InvisibleFakeTorso.Parent = workspace
        Camera.CameraType = Enum.CameraType.Custom
        Camera.CameraSubject = _G.MiscState.InvisibleFakeTorso

        if not silent then
            RunService:BindToRenderStep("MeditationFocus", Enum.RenderPriority.Camera.Value - 1, function()
                if not _G.MiscState.IsInvisible or not hrp or not hrp.Parent or not _G.MiscState.InvisibleFakeTorso then return end
                local camCF = Camera.CFrame
                local lookDir = Vector3.new(camCF.LookVector.X, 0, camCF.LookVector.Z)
                if lookDir.Magnitude > 0.001 then
                    hrp.CFrame = CFrame.new(hrp.Position, hrp.Position + lookDir.Unit)
                end
                _G.MiscState.InvisibleFakeTorso.CFrame = CFrame.new(hrp.Position + Vector3.new(0, 1.5, 0))
            end)
        end

        local animObj = ReplicatedStorage.Modules.MVP.Meditation.Character
        _G.MiscState.InvisibleTrack = animator:LoadAnimation(animObj)
        _G.MiscState.InvisibleTrack.Priority = Enum.AnimationPriority.Action4
        _G.MiscState.InvisibleTrack:Play()
        task.wait(0.1)
        _G.MiscState.InvisibleTrack.TimePosition = 0.1
        _G.MiscState.InvisibleTrack:AdjustSpeed(0)

        for _, t in ipairs(animator:GetPlayingAnimationTracks()) do
            if t ~= _G.MiscState.InvisibleTrack then t:Stop(0) end
        end

        Invisibility.Connections["InvisAnim"] = animator.AnimationPlayed:Connect(function(newTrack)
            if newTrack ~= _G.MiscState.InvisibleTrack then newTrack:Stop(0) end
        end)
        library:create_notification({name = "Invisibility Enabled", duration = 2})
    else
        _G.MiscState.IsInvisible = false
        
        if Invisibility.Connections["InvisNoclip"] then 
            Invisibility.Connections["InvisNoclip"]:Disconnect() 
            Invisibility.Connections["InvisNoclip"] = nil 
        end
        if Invisibility.Connections["InvisAnim"] then 
            Invisibility.Connections["InvisAnim"]:Disconnect() 
            Invisibility.Connections["InvisAnim"] = nil 
        end
        pcall(function() RunService:UnbindFromRenderStep("MeditationFocus") end)

        if char and char.Parent then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
        if hrp and hrp.Parent then hrp.Anchored = false end
        if hum and hum.Parent then hum.AutoRotate = true end
        if hum and hum.Parent then Camera.CameraSubject = hum end
        if _G.MiscState.InvisibleFakeTorso then 
            _G.MiscState.InvisibleFakeTorso:Destroy() 
            _G.MiscState.InvisibleFakeTorso = nil 
        end
        if _G.MiscState.InvisibleTrack then
            pcall(function() _G.MiscState.InvisibleTrack:Stop(0) end)
            pcall(function() _G.MiscState.InvisibleTrack:Destroy() end)
            _G.MiscState.InvisibleTrack = nil
        end
        library:create_notification({name = "Invisibility Disabled", duration = 2})
    end
end

local function toggleInfiniteDash(state)
    _G.MiscState.InfiniteDashActive = state
    
    -- Clean up any existing connection
    if _G.MiscState.InfDashConn then
        _G.MiscState.InfDashConn:Disconnect()
        _G.MiscState.InfDashConn = nil
    end
    
    if state then
        -- Safely grab the game's Movement Controller
        local ok, module = pcall(function() 
            return require(LocalPlayer.PlayerScripts.Controllers.Character.MovementController) 
        end)
        
        if not ok or type(module) ~= "table" or type(module.Dash) ~= "function" then 
            if library and library.create_notification then
                library:create_notification({name = "Dash module not found!, don't use xeno", duration = 3})
            end
            return 
        end
        
        -- Constantly reset the internal cooldown variables silently
        _G.MiscState.InfDashConn = RunService.Heartbeat:Connect(function()
            if not _G.MiscState.InfiniteDashActive then return end
            pcall(function()
                -- Scan the Dash function's hidden variables (upvalues)
                for i, val in pairs(debug.getupvalues(module.Dash)) do
                    -- If it's a number (cooldowns and timestamps are numbers), force it to 0
                    if type(val) == "number" then 
                        debug.setupvalue(module.Dash, i, 0) 
                    end
                end
            end)
        end)
    end
end

local function EnsureACBypassed(actionName, toggleUIElement, callback)
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end

    -- Reset cache if it's a new life
    if _G.StandState.CurrentLifeChar ~= char then
        _G.StandState.CurrentLifeChar = char
        _G.StandState.IsACVerified = nil
    end

    if _G.StandState.IsACVerified == true then
        task.spawn(callback)
        return
    elseif _G.StandState.IsACVerified == false then
        library:create_notification({name = "AC Check Failed! Reset character to try again.", duration = 3})
        if toggleUIElement then pcall(function() toggleUIElement:set(false) end) end
        return
    end

    if _G.StandState.IsCheckingAC then
        library:create_notification({name = "Currently verifying AC...", duration = 2})
        if toggleUIElement then pcall(function() toggleUIElement:set(false) end) end
        return
    end

    task.spawn(function()
        _G.StandState.IsCheckingAC = true
        library:create_notification({name = "Quick AC Check...", duration = 2})
        
        local root = char.HumanoidRootPart
        local originalCFrame = root.CFrame
        local testCoord = Vector3.new(-30.5004654, 23.4476242, -65.0752106)
        
        -- Teleport to test spot
        root.CFrame = CFrame.new(testCoord)
        task.wait(0.35) 
        
        local currentCFrame = root.CFrame
        
        -- Check for fluke/rubberband (if we are close to original CFrame, we rubberbanded)
        local rubberbanded = (currentCFrame.Position - originalCFrame.Position).Magnitude < 15
        local stayedAtTest = (currentCFrame.Position - testCoord).Magnitude < 15

        if rubberbanded or not stayedAtTest then
            _G.StandState.IsACVerified = false
            library:create_notification({name = actionName .. " Cancelled: AC Not Bypassed!", duration = 3})
            if toggleUIElement then pcall(function() toggleUIElement:set(false) end) end
            pcall(function() root.CFrame = originalCFrame end)
        else
            _G.StandState.IsACVerified = true
            library:create_notification({name = "AC Verified! Executing " .. actionName, duration = 2})
            pcall(function() root.CFrame = originalCFrame end)
            task.wait(0.1)
            callback()
        end
        
        _G.StandState.IsCheckingAC = false
    end)
end

local function GetTargetByPartialName(str)
    if not str or str == "" then return nil end
    str = string.lower(str)
    for _, p in ipairs(Players:GetPlayers()) do
        if string.sub(string.lower(p.Name), 1, #str) == str or string.sub(string.lower(p.DisplayName), 1, #str) == str then
            if p.Character then return p.Character end
        end
    end
    local charsFolder = workspace:FindFirstChild("Characters")
    if charsFolder then
        for _, obj in ipairs(charsFolder:GetChildren()) do
            if string.sub(string.lower(obj.Name), 1, #str) == str then
                return obj
            end
        end
    end
    return nil
end

local function GetClosestTargetInFront(originChar, maxRadius, maxAngle)
    local originRoot = originChar:FindFirstChild("HumanoidRootPart")
    if not originRoot then return nil end
    local closestTarget = nil
    local minDistance = maxRadius
    
    local function checkTarget(t)
        if t ~= originChar and t ~= LocalPlayer.Character then
            local tRoot = t:FindFirstChild("HumanoidRootPart")
            local tHum = t:FindFirstChild("Humanoid")
            if tRoot and tHum and tHum.Health > 0 then
                local dir = tRoot.Position - originRoot.Position
                local dist = dir.Magnitude
                if dist <= maxRadius then
                    local dot = originRoot.CFrame.LookVector:Dot(dir.Unit)
                    local angle = math.deg(math.acos(dot))
                    if angle <= maxAngle then
                        if dist < minDistance then
                            minDistance = dist
                            closestTarget = t
                        end
                    end
                end
            end
        end
    end
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then checkTarget(p.Character) end
    end
    local charsFolder = workspace:FindFirstChild("Characters")
    if charsFolder then
        for _, m in ipairs(charsFolder:GetChildren()) do
            if m:IsA("Model") then checkTarget(m) end
        end
    end
    return closestTarget
end

            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end)

    local animator = humanoid:FindFirstChildOfClass("Animator")
    if not animator then
        animator = Instance.new("Animator")
        animator.Parent = humanoid
    end

    -- Old Stand Animation (Idle)
    local idleAnimation = Instance.new("Animation")
    idleAnimation.AnimationId = "rbxassetid://89028642789375"
    _G.StandState.IdleTrack = animator:LoadAnimation(idleAnimation)
    _G.StandState.IdleTrack.Priority = Enum.AnimationPriority.Action4

    -- New Stand Animation (Rise & Move)
    local moveAnimation = Instance.new("Animation")
    moveAnimation.AnimationId = "rbxassetid://129715807085749"
    _G.StandState.MoveTrack = animator:LoadAnimation(moveAnimation)
    _G.StandState.MoveTrack.Priority = Enum.AnimationPriority.Action4

    -- Play Rise Animation and Freeze at 0.58
    _G.StandState.MoveTrack:Play()
    task.wait(0.05)
    _G.StandState.MoveTrack:AdjustSpeed(0)
    _G.StandState.MoveTrack.TimePosition = 0.58

    local moveSpeed = math.max(_G.StandState.RiseSpeed or 50, 10)
    local riseDistance = 25 
    local riseDuration = math.clamp(riseDistance / moveSpeed, 0.2, 5)
    local elapsed = 0
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = { character, targetChar }

    while _G.StandState.IsAttaching and elapsed < riseDuration do
        if not targetChar or not targetRoot or not targetRoot.Parent then break end
        local delta = RunService.Heartbeat:Wait()
        elapsed = elapsed + delta
        
        local alpha = math.clamp(elapsed / riseDuration, 0, 1)
        alpha = 1 - math.pow(1 - alpha, 3) 

        local rightVector = targetRoot.CFrame.RightVector
        local lookVector = targetRoot.CFrame.LookVector
        local offsetXZ = (rightVector * 2.5) - (lookVector * 2.5)
        local desiredPosXZ = targetRoot.Position + offsetXZ
        
        local groundY = targetRoot.Position.Y - 3
        local rayResult = workspace:Raycast(targetRoot.Position, Vector3.new(0, -15, 0), raycastParams)
        if rayResult then groundY = rayResult.Position.Y end
        
        local targetY = groundY + 5
        local startY = groundY - 20
        
        local currentY = startY + ((targetY - startY) * alpha)
        local currentPos = Vector3.new(desiredPosXZ.X, currentY, desiredPosXZ.Z)
        
        root.CFrame = CFrame.new(currentPos) * targetRoot.CFrame.Rotation
    end
    
    if not _G.StandState.IsAttaching then return end

    ApplyCameraSubject()

    -- SUMMON DONE: Turn ON Collisions (Stop Noclip loop)
    if _G.StandState.Conns.Noclip then
        _G.StandState.Conns.Noclip:Disconnect()
        _G.StandState.Conns.Noclip = nil
    end
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = true end
    end

    -- Switch to Old Idle Animation and Freeze at 0.7
    _G.StandState.MoveTrack:Stop()
    _G.StandState.IdleTrack:Play()
    task.wait(0.05)
    _G.StandState.IdleTrack:AdjustSpeed(0)
    _G.StandState.IdleTrack.TimePosition = 0.7

    local wasMoving = false

    -- Create local connection variable so it can self-terminate
    local renderConn 
    renderConn = RunService.RenderStepped:Connect(function()
        -- SELF TERMINATION CHECK (Fixes the race condition bug)
        if not _G.StandState.IsAttaching then
            if renderConn then renderConn:Disconnect() end
            return
        end

        if _G.StandState.IsPerformingAction then return end 
        if not targetChar or not targetChar.Parent or not targetRoot or not targetRoot.Parent then return end

        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero

        -- Check if Master is moving
        local isMoving = false
        local tHum = targetChar:FindFirstChild("Humanoid")
        if tHum and tHum.MoveDirection.Magnitude > 0 then
            isMoving = true
        end

        -- Handle Animation States
        if isMoving and not wasMoving then
            wasMoving = true
            _G.StandState.IdleTrack:Stop()
            _G.StandState.MoveTrack:Play()
            _G.StandState.MoveTrack:AdjustSpeed(0)
            _G.StandState.MoveTrack.TimePosition = 3.55
        elseif not isMoving and wasMoving then
            wasMoving = false
            _G.StandState.MoveTrack:Stop()
            _G.StandState.IdleTrack:Play()
            _G.StandState.IdleTrack:AdjustSpeed(0)
            _G.StandState.IdleTrack.TimePosition = 0.7
        end

        if _G.StandState.IsAuraActive then
            if _G.StandState.AuraCurrentEnemy and _G.StandState.AuraCurrentEnemy:FindFirstChild("HumanoidRootPart") then
                local enemyRoot = _G.StandState.AuraCurrentEnemy.HumanoidRootPart
                local desiredPos = enemyRoot.Position + Vector3.new(0, -5, 0)
                root.CFrame = CFrame.lookAt(desiredPos, enemyRoot.Position)
            else
                local orbitSpeed = 3
                local orbitRadius = 7
                local angle = tick() * orbitSpeed
                local offsetX = math.cos(angle) * orbitRadius
                local offsetZ = math.sin(angle) * orbitRadius
                
                local desiredPosXZ = targetRoot.Position + Vector3.new(offsetX, 0, offsetZ)
                local groundY = targetRoot.Position.Y - 3
                local rayResult = workspace:Raycast(desiredPosXZ, Vector3.new(0, -15, 0), raycastParams)
                if rayResult then groundY = rayResult.Position.Y end
                
                local floatOffset = math.sin(tick() * 1.5) * 0.6
                local finalY = groundY + 5 + floatOffset
                local finalPosition = Vector3.new(desiredPosXZ.X, finalY, desiredPosXZ.Z)
                root.CFrame = CFrame.lookAt(finalPosition, targetRoot.Position)
            end
        else
            local rightVector = targetRoot.CFrame.RightVector
            local lookVector = targetRoot.CFrame.LookVector
            local offsetXZ = (rightVector * 2.5) - (lookVector * 2.5)
            local desiredPosXZ = targetRoot.Position + offsetXZ

            local groundY = targetRoot.Position.Y - 3
            local rayResult = workspace:Raycast(targetRoot.Position, Vector3.new(0, -15, 0), raycastParams)
            if rayResult then groundY = rayResult.Position.Y end

            local floatOffset = math.sin(tick() * 1.5) * 0.6
            local finalY = groundY + 5 + floatOffset
            local finalPosition = Vector3.new(desiredPosXZ.X, finalY, desiredPosXZ.Z)

            root.CFrame = CFrame.new(finalPosition) * targetRoot.CFrame.Rotation
        end
    end)
    
    _G.StandState.Conns.Render = renderConn
end

local function PerformAttackSequence(keyString, originChar)
    local enemyChar = GetClosestTargetInFront(originChar, 100, 60)
    if not enemyChar then return end
    
    _G.StandState.IsPerformingAction = true
    local root = LocalPlayer.Character.HumanoidRootPart
    local enemyRoot = enemyChar.HumanoidRootPart
    
    root.Anchored = true
    local offsetXZ = enemyRoot.CFrame.LookVector * 3.5
    local targetPos = enemyRoot.Position + offsetXZ
    
    local tInfo = TweenInfo.new((root.Position - targetPos).Magnitude / 200, Enum.EasingStyle.Linear)
    local tween = TweenService:Create(root, tInfo, {CFrame = CFrame.lookAt(targetPos, enemyRoot.Position)})
    tween:Play()
    tween.Completed:Wait()
    
    local attackRenderConn = RunService.RenderStepped:Connect(function()
        if not enemyRoot or not enemyRoot.Parent then return end
        local frontPos = enemyRoot.Position + (enemyRoot.CFrame.LookVector * 3.5)
        root.CFrame = CFrame.lookAt(frontPos, enemyRoot.Position)
    end)
    
    local KeycodeMap = { ["1"] = Enum.KeyCode.One, ["2"] = Enum.KeyCode.Two, ["3"] = Enum.KeyCode.Three, ["4"] = Enum.KeyCode.Four }
    local animator = LocalPlayer.Character.Humanoid:FindFirstChildOfClass("Animator")
    local detectedAnim = nil
    local animConn = animator.AnimationPlayed:Connect(function(track) detectedAnim = track end)
    
    VirtualInputManager:SendKeyEvent(true, KeycodeMap[keyString], false, game)
    task.wait(0.05)
    VirtualInputManager:SendKeyEvent(false, KeycodeMap[keyString], false, game)
    
    local t = 0
    while t < 1 and not detectedAnim do t = t + task.wait(0.1) end
    animConn:Disconnect()
    
    if detectedAnim then
        local stopped = false
        local stopConn = detectedAnim.Stopped:Connect(function() stopped = true end)
        local timeout = 0
        while not stopped and timeout < 5 do timeout = timeout + task.wait(0.1) end
        if stopConn then stopConn:Disconnect() end
    else
        task.wait(1.5) 
    end
    
    if attackRenderConn then attackRenderConn:Disconnect() end
    
    local standRoot = originChar:FindFirstChild("HumanoidRootPart")
    if standRoot then
        local sRight = standRoot.CFrame.RightVector
        local sLook = standRoot.CFrame.LookVector
        local sOffsetXZ = (sRight * 2.5) - (sLook * 2.5)
        local sTargetPos = standRoot.Position + sOffsetXZ
        local tInfo2 = TweenInfo.new((root.Position - sTargetPos).Magnitude / 200, Enum.EasingStyle.Linear)
        local tween2 = TweenService:Create(root, tInfo2, {CFrame = CFrame.lookAt(sTargetPos, standRoot.Position)})
        tween2:Play()
        tween2.Completed:Wait()
    end
    
    root.Anchored = false
    _G.StandState.IsPerformingAction = false
    
    if _G.StandState.IntendedState == "Attached" and not _G.StandState.IsAttaching then
        _G.StandState.IsAttaching = true
        task.spawn(function() AttachToTarget(originChar) end)
    end
end

local function ExecuteBringLogic(PlayerToBring, StandUserChar)
    local MainTargetHrp = StandUserChar:FindFirstChild("HumanoidRootPart")
    if not MainTargetHrp or not PlayerToBring or not PlayerToBring:FindFirstChild("HumanoidRootPart") then return end
    
    local charModel = LocalPlayer.Character
    if not charModel then return end
    
    local currentMoveset = charModel:GetAttribute("Moveset")
    
    -- Dictionary holding supported movesets, their respective remotes, move names, and wait delays
    local supportedMovesets = {
        ["Itadori"] = { Service = "CursedStrikesService", MoveName = "Cursed Strikes", Delay = 0.55 },
        ["Gojo"] = { Service = "RapidPunchesService", MoveName = "Rapid Punches", Delay = 0.55 },
        ["Choso"] = { Service = "BloodEdgeService", MoveName = "Blood Edge", Delay = 0.55 },
        ["Todo"] = { Service = "SwiftKickService", MoveName = "Swift Kick", Delay = 1.50 }, -- Todo stays behind for 1.5s
        ["Mechamaru"] = { Service = "BoostOnService", MoveName = "Boost On", Delay = 0.55 },
        ["Kurourushi"] = { Service = "ChokeholdService", MoveName = "Chokehold", Delay = 0.55 },
        ["Yuta"] = { Service = "SecondWindService", MoveName = "Second Wind", Delay = 0.55 }
    }
    
    local config = supportedMovesets[currentMoveset]
    
    if not config then
        library:create_notification({name = "Bring failed: '" .. tostring(currentMoveset) .. "' is not supported!", duration = 3})
        return
    end
    
    local movesetFolder = charModel:FindFirstChild("Moveset")
    local moveInstance = movesetFolder and movesetFolder:FindFirstChild(config.MoveName)
    
    if not moveInstance then
        library:create_notification({name = "Move '" .. config.MoveName .. "' is currently on cooldown or missing!", duration = 3})
        return
    end
    
    local hrp = charModel:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    _G.StandState.IsPerformingAction = true
    hrp.Anchored = true 
    
    local targetToBringHrp = PlayerToBring.HumanoidRootPart
    
    local trackConn = RunService.RenderStepped:Connect(function()
        if not targetToBringHrp or not targetToBringHrp.Parent then return end
        -- Teleport 1 stud behind the target's back (Positive Z is backwards relative to LookVector)
        local behindPos = (targetToBringHrp.CFrame * CFrame.new(0, 0, 1)).Position
        -- Lock body to perfectly face the target's back
        hrp.CFrame = CFrame.lookAt(behindPos, targetToBringHrp.Position)
    end)
    
    task.wait(0.1)
    
    -- Fire the specific RemoteEvent associated with the current character
    pcall(function()
        local remote = ReplicatedStorage.Knit.Knit.Services[config.Service].RE.Activated
        if currentMoveset == "Yuta" then
            remote:FireServer(moveInstance, nil)
        else
            remote:FireServer(moveInstance)
        end
    end)
    
    -- Wait based on the moveset's specific delay time (0.55s default, 1.5s for Todo)
    task.wait(config.Delay)
    
    if trackConn then trackConn:Disconnect() end
    
    -- ==========================================
    -- SAFE DROP-OFF LOGIC (8 Studs Above Master)
    -- ==========================================
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    
    -- Filter out Stand, Master, and Victim so the raycast only hits map ceilings/roofs
    local filterList = {charModel, StandUserChar}
    if PlayerToBring.Parent then table.insert(filterList, PlayerToBring.Parent) end
    raycastParams.FilterDescendantsInstances = filterList

    local masterPos = MainTargetHrp.Position
    local upCheck = Vector3.new(0, 8, 0)
    local safeHeight = 8
    
    -- Cast a ray straight up from the Master
    local rayResult = workspace:Raycast(masterPos, upCheck, raycastParams)
    if rayResult then
        -- If we hit a ceiling, calculate the distance. Subtract 3.5 studs so the character's head doesn't clip into the roof
        safeHeight = math.max(0, (rayResult.Position.Y - masterPos.Y) - 3.5)
    end
    
    local dropPos = masterPos + Vector3.new(0, safeHeight, 0)
    -- Teleport above the master, facing the exact same direction the master is facing
    hrp.CFrame = CFrame.lookAt(dropPos, dropPos + MainTargetHrp.CFrame.LookVector)
    
    task.wait(0.5)
    
    -- ==========================================
    -- RETURN TO IDLE POSITION
    -- ==========================================
    -- Tween back into the perfect side-idle position next to the Master
    local sRight = MainTargetHrp.CFrame.RightVector
    local sLook = MainTargetHrp.CFrame.LookVector
    local sOffsetXZ = (sRight * 2.5) - (sLook * 2.5)
    local sTargetPos = MainTargetHrp.Position + sOffsetXZ
    
    local tInfo2 = TweenInfo.new((hrp.Position - sTargetPos).Magnitude / 200, Enum.EasingStyle.Linear)
    local tween2 = TweenService:Create(hrp, tInfo2, {CFrame = CFrame.lookAt(sTargetPos, MainTargetHrp.Position)})
    tween2:Play()
    tween2.Completed:Wait()
    
    hrp.Anchored = false
    _G.StandState.IsPerformingAction = false
    
    -- Automatically re-attach if we were attached
    if _G.StandState.IntendedState == "Attached" and not _G.StandState.IsAttaching then
        _G.StandState.IsAttaching = true
        task.spawn(function() AttachToTarget(StandUserChar) end)
    end
end

local function StartAuraLoop()
    _G.StandState.Threads.Aura = task.spawn(function()
        while _G.StandState.IsAuraActive and _G.StandState.IsAttaching do
            local mainHrp = _G.StandState.MainTarget and _G.StandState.MainTarget:FindFirstChild("HumanoidRootPart")
            if not mainHrp then task.wait(0.1) continue end
            
            local enemiesInRange = {}
            local auraRadius = 30
            
            local function checkAuraTarget(t)
                if t ~= _G.StandState.MainTarget and t ~= LocalPlayer.Character then
                    local tHrp = t:FindFirstChild("HumanoidRootPart")
                    local tHum = t:FindFirstChild("Humanoid")
                    if tHrp and tHum and tHum.Health > 0 then
                        local dist = (tHrp.Position - mainHrp.Position).Magnitude
                        if dist <= auraRadius then
                            table.insert(enemiesInRange, t)
                        end
                    end
                end
            end
            
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Character then checkAuraTarget(p.Character) end
            end
            local charsFolder = workspace:FindFirstChild("Characters")
            if charsFolder then
                for _, m in ipairs(charsFolder:GetChildren()) do
                    if m:IsA("Model") then checkAuraTarget(m) end
                end
            end
            
            if #enemiesInRange > 0 then
                for _, enemy in ipairs(enemiesInRange) do
                    if not _G.StandState.IsAuraActive or not _G.StandState.IsAttaching then break end
                    if enemy and enemy:FindFirstChild("HumanoidRootPart") then
                        _G.StandState.AuraCurrentEnemy = enemy
                        task.wait(0.1) 
                        for i = 1, 3 do 
                            if not _G.StandState.IsAuraActive or not _G.StandState.IsAttaching or not enemy:FindFirstChild("HumanoidRootPart") then break end
                            firedynamicm1()
                            task.wait(0.2)
                        end
                    end
                end
                _G.StandState.AuraCurrentEnemy = nil
                task.wait(0.2) 
            else
                task.wait(0.1)
            end
        end
    end)
end

local function HandleAutoSoda()
    if _G.StandState.IsGettingSoda then return end
    
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChild("Humanoid")
    
    if not hum or hum.Health >= 30 then return end
    
    _G.StandState.IsGettingSoda = true
    _G.StandState.IsPerformingAction = true -- Temporarily suspends the Stand's attachment to Master
    
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local shopPos = CFrame.new(247.931198, 23.0549088, 593.589783)
    
    if _G.StandState.AutoSoda and hrp and hum and hum.Health > 0 then
        
        -- 1. Robust TP to Shop (Verify we reached the spot)
        local reachedShop = false
        for i = 1, 10 do 
            if not hrp or not hrp.Parent then break end
            hrp.CFrame = shopPos
            task.wait(0.3)
            
            if (hrp.Position - shopPos.Position).Magnitude < 10 then
                reachedShop = true
                break
            end
        end
        
        if not reachedShop then 
            _G.StandState.IsPerformingAction = false
            _G.StandState.IsGettingSoda = false
            return 
        end
        
        local itemsFolder = workspace:WaitForChild("Items", 3)
        local foundPrompt, foundPart = nil, nil
        
        -- 2. Purchase Verification Loop
        for attempt = 1, 5 do 
            if not _G.StandState.AutoSoda or not hrp or not hrp.Parent then break end
            
            hrp.CFrame = shopPos
            
            pcall(function()
                ReplicatedStorage.Knit.Knit.Services.ShopService.RE.PurchaseSoda:FireServer()
            end)
            
            local waitStart = tick()
            while tick() - waitStart < 5 do
                if itemsFolder then
                    for _, item in ipairs(itemsFolder:GetChildren()) do
                        if string.find(item.Name, "Soda") then
                            local part = item:IsA("BasePart") and item or item:FindFirstChildWhichIsA("BasePart", true)
                            local prompt = item:FindFirstChildWhichIsA("ProximityPrompt", true)
                            
                            if part and prompt and (part.Position - shopPos.Position).Magnitude < 30 then
                                foundPart = part
                                foundPrompt = prompt
                                break
                            end
                        end
                    end
                end
                if foundPart then break end
                task.wait(0.2)
            end
            
            if foundPart then break end 
        end
        
        if foundPart and foundPrompt then
            -- 3. Simple TP exactly to the Soda Part
            hrp.CFrame = CFrame.new(foundPart.Position)
            task.wait(0.3)
            
            -- 4. Exploit prompt duration to 0 and fire it instantly
            pcall(function()
                foundPrompt.HoldDuration = 0
                if fireproximityprompt then
                    fireproximityprompt(foundPrompt)
                end
            end)
            
            task.wait(0.5) 
            
            -- 5. Drink it (M1 Remote)
            pcall(function()
                ReplicatedStorage.Knit.Knit.Services.ItemService.RE.M1:FireServer(false)
            end)
            
            -- 6. Wait exactly 5 seconds after drinking
            task.wait(5)
        end
        
        -- 7. Turn off the toggle in the UI, then Re-arm after 5 seconds
        _G.StandState.ScriptDisabledSoda = true
        if _G.StandToggles and _G.StandToggles.AutoSoda then
            _G.StandToggles.AutoSoda:set(false)
        else
            _G.StandState.AutoSoda = false
        end
        
        task.spawn(function()
            task.wait(5)
            -- Only turn it back on if the player hasn't manually interfered
            if _G.StandState.ScriptDisabledSoda then
                _G.StandState.ScriptDisabledSoda = false
                if _G.StandToggles and _G.StandToggles.AutoSoda then
                    _G.StandToggles.AutoSoda:set(true)
                else
                    _G.StandState.AutoSoda = true
                end
            end
        end)
    end
    
    -- 8. Simple TP back to Master (With verification)
    local finalChar = LocalPlayer.Character
    local finalHrp = finalChar and finalChar:FindFirstChild("HumanoidRootPart")
    
    if _G.StandState.MainTarget and finalHrp then
        local targetRoot = _G.StandState.MainTarget:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            for i = 1, 5 do
                if not finalHrp or not finalHrp.Parent or not targetRoot.Parent then break end
                finalHrp.CFrame = CFrame.new(targetRoot.Position)
                task.wait(0.2)
                if (finalHrp.Position - targetRoot.Position).Magnitude < 20 then
                    break
                end
            end
        end
    end
    
    _G.StandState.IsPerformingAction = false
    _G.StandState.IsGettingSoda = false
end

            if child:IsA("Sound") and child.Name == "Death" then

                        EnsureACBypassed("Sound Move", nil, function()
                            task.spawn(function() PerformAttackSequence(action, hrp.Parent) end)
                        end)
                    end
                elseif action == "bring" then
                    if _G.StandState.IsAttaching and not _G.StandState.IsPerformingAction then
                        local enemyChar = GetClosestTargetInFront(hrp.Parent, 100, 60)
                        if enemyChar then
                            EnsureACBypassed("Sound Bring", nil, function()
                                task.spawn(function() ExecuteBringLogic(enemyChar, hrp.Parent) end)
                            end)
                        end
                    end
                elseif action == "abuse" then
                    local enemyChar = GetClosestTargetInFront(hrp.Parent, 100, 60)
                    if enemyChar then
                        UpdateBringTarget(enemyChar.Name)
                        if _G.StandToggles.Abuse then _G.StandToggles.Abuse:set(true) end
                        library:create_notification({name = "Stand abuse mode auto-triggered!", duration = 2})
                    end
                end
            end
        end)
    end
    
    while _G.StandState.IsMonitoringSound do
        local char = getWorkspaceChar(targetName)
        if char then
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp and hrp ~= monitoredHrp then
                monitoredHrp = hrp
                HookRootPart(hrp)
            end
        end
        task.wait(1)
    end
    
    if hrpChildAddedConn then hrpChildAddedConn:Disconnect() end
end

local function OnPlayerChatted(player, message)
    local msg = string.lower(message)
    if string.sub(msg, 1, 3) ~= ".s " then return end

    local args = string.split(msg, " ")
    local cmd = args[2]
    local argName = args[3]

    local isOriginalMaster = (_G.StandState.OriginalMasterName and string.lower(player.Name) == string.lower(_G.StandState.OriginalMasterName))
    local isCurrentMaster  = (_G.StandState.CurrentMasterName and string.lower(player.Name) == string.lower(_G.StandState.CurrentMasterName))

    if cmd == "retrieve" and isOriginalMaster then
        local tChar = getWorkspaceChar(_G.StandState.OriginalMasterName)
        if tChar then
            UpdateMainTarget(_G.StandState.OriginalMasterName)
            if _G.StandToggles.Attach then _G.StandToggles.Attach:set(true) end
        end
        return
    end

    if not isCurrentMaster then return end

    if cmd == "bring" and argName then
        local targetToBring = GetTargetByPartialName(argName)
        if targetToBring and _G.StandState.MainTarget then
            UpdateBringTarget(targetToBring.Name)
            EnsureACBypassed("Chat Bring", nil, function()
                task.spawn(function() ExecuteBringLogic(targetToBring, _G.StandState.MainTarget) end)
            end)
        end
    elseif cmd == "abuse" and argName then
        local targetToAbuse = GetTargetByPartialName(argName)
        if targetToAbuse then
            UpdateBringTarget(targetToAbuse.Name)
            if _G.StandToggles.Abuse then _G.StandToggles.Abuse:set(true) end
        end
    elseif cmd == "aura" then
        if _G.StandState.IsAttaching then
            if _G.StandToggles.Aura then 
                _G.StandToggles.Aura:set(not _G.StandState.IsAuraActive) 
            end
        end
    elseif cmd == "givestand" and argName then
        local newMasterChar = GetTargetByPartialName(argName)
        if newMasterChar then
            _G.StandState.CurrentMasterName = newMasterChar.Name
            UpdateMainTarget(newMasterChar.Name)
            if _G.StandToggles.Attach then _G.StandToggles.Attach:set(true) end
        end
    elseif cmd == "move" and argName then
        if (argName == "1" or argName == "2" or argName == "3" or argName == "4") and _G.StandState.IsAttaching then
            EnsureACBypassed("Chat Move", nil, function()
                task.spawn(function() PerformAttackSequence(argName, _G.StandState.MainTarget) end)
            end)
        end
    elseif cmd == "unsummon" then
        if _G.StandToggles.Attach then _G.StandToggles.Attach:set(false) end
        if _G.StandToggles.Abuse then _G.StandToggles.Abuse:set(false) end
        if _G.StandToggles.Aura then _G.StandToggles.Aura:set(false) end
    elseif cmd == "summon" then
        if _G.StandState.MainTarget then
            if _G.StandToggles.Attach then _G.StandToggles.Attach:set(true) end
        end
    end
end

local function HookChat(player)
    player.Chatted:Connect(function(message)
        OnPlayerChatted(player, message)
    end)
end

for _, p in ipairs(Players:GetPlayers()) do HookChat(p) end
Players.PlayerAdded:Connect(HookChat)

-- Dynamic test to detect absurd domain coordinates
local function isAbsurdLocation(pos)
    -- Known domain location cluster center (ignoring Y axis height differences)
    local flatPos = Vector2.new(pos.X, pos.Z)
    local domainCenter = Vector2.new(2000, -3000)
    
    -- If they are within a 1500 stud radius of the 2000, -3000 cluster
    if (flatPos - domainCenter).Magnitude < 1500 then
        return true
    end
    
    -- Generic fallback: if they are further than 3500 studs from the center of the world (0,0,0)
    if pos.Magnitude > 3500 then
        return true
    end
    
    return false
end

local function runGojo02Kill()
    if _G.AutoKillState.IsGojo02Active then
        library:create_notification({name = "Gojo sequence already running!", duration = 2})
        return
    end
    
    local char = LocalPlayer.Character
    if not char then return end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    
    -- NEW: Check if Ultimate is ready using attributes
    local ultCharge = LocalPlayer:GetAttribute("Ultimate") or 0
    if ultCharge < 100 then
        library:create_notification({name = "you don't have ultimate", duration = 3})
        return
    end
    
    _G.AutoKillState.IsGojo02Active = true
    
    local startTime = tick()
    
    -- 1. Fire Ultimate Remote
    pcall(function()
        game:GetService("ReplicatedStorage").Knit.Knit.Services.GojoService.RE.Ultimate:FireServer()
    end)
    
    -- 2. Wait EXACTLY 0.25 seconds (Crucial timing window)
    task.wait(0.25)
    
    -- 3. Fire RightActivated remote 3 times with a 0.01s interval
    for i = 1, 3 do
        pcall(function()
            local args = { [1] = nil, n = 1 }
            game:GetService("ReplicatedStorage").Knit.Knit.Services.GojoService.RE.RightActivated:FireServer(unpack(args, 1, args.n or #args))
        end)
        if i < 3 then 
            task.wait(0.01) 
        end
    end
    
    -- 4. Wait until 6 seconds total has passed since the ultimate was fired
    local timePassed = tick() - startTime
    local timeRemaining = 6 - timePassed
    if timeRemaining > 0 then
        task.wait(timeRemaining)
    end
    
    -- Verify the player hasn't died during the waiting period
    if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") or (LocalPlayer.Character:FindFirstChild("Humanoid") and LocalPlayer.Character.Humanoid.Health <= 0) then
        _G.AutoKillState.IsGojo02Active = false
        return
    end
    
    local mode = _G.AutoKillState.GojoMode
    local useInvis = (mode == "invis + teleport" or mode == "tp + invis + die at end")
    local dieAtEnd = (mode == "tp + die at end" or mode == "tp + invis + die at end")
    
    if useInvis then
        pcall(function() Invisibility.toggle(true) end)
    end
    
    -- 5. Start Teleporting (Duration: 7 seconds to reach the 13 second mark)
    local endTime = tick() + 7
    while tick() < endTime and _G.AutoKillState.IsGojo02Active do
        local targetPlayers = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local tHum = p.Character:FindFirstChildOfClass("Humanoid")
                local tRoot = p.Character:FindFirstChild("HumanoidRootPart")
                if tHum and tHum.Health > 0 and tRoot then
                    
                    local skipPlayer = false
                    
                    -- Check if they are inside a domain / absurd location
                    if _G.AutoKillState.ExcludeDomains and isAbsurdLocation(tRoot.Position) then
                        skipPlayer = true
                    end
                    
                    -- Check if they are in the excluded players list
                    if _G.AutoKillState.ExcludePlayersToggle and type(_G.AutoKillState.ExcludedPlayers) == "table" then
                        for k, v in pairs(_G.AutoKillState.ExcludedPlayers) do
                            local exName = type(k) == "string" and k or v 
                            if p.Name == exName or p.DisplayName == exName then
                                skipPlayer = true
                                break
                            end
                        end
                    end
                    
                    if not skipPlayer then
                        table.insert(targetPlayers, {player = p, hp = tHum.Health})
                    end
                end
            end
        end
        
        if mode == "Teleport to low hp player first" then
            table.sort(targetPlayers, function(a, b) return a.hp < b.hp end)
        end
        
        for _, data in ipairs(targetPlayers) do
            if tick() >= endTime or not _G.AutoKillState.IsGojo02Active then break end
            
            local localChar = LocalPlayer.Character
            local tChar = data.player.Character
            
            if localChar and tChar then
                local localRoot = localChar:FindFirstChild("HumanoidRootPart")
                local tRoot = tChar:FindFirstChild("HumanoidRootPart")
                if localRoot and tRoot then
                    localRoot.CFrame = tRoot.CFrame * CFrame.new(0, 0, 3)
                end
            end
            task.wait(_G.AutoKillState.GojoStayTime)
        end
        
        if #targetPlayers == 0 then
            task.wait(0.1)
        end
    end
    
    -- 6. Sequence Over (13 Seconds Total)
    if useInvis then
        pcall(function() Invisibility.toggle(false) end)
    end
    
    if dieAtEnd then
        local localRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if localRoot then
            localRoot.CFrame = CFrame.new(localRoot.Position.X, -200, localRoot.Position.Z)

    local camera = workspace.CurrentCamera
    
    if not localHrp then return end

            remote:FireServer(targetModel)
        end)

        localHrp.Anchored = false
        
        localHrp.CFrame = originalCFrame
        localHrp.AssemblyLinearVelocity = Vector3.zero
        localHrp.AssemblyAngularVelocity = Vector3.zero
        
        task.wait(0.7)
        
        localHrp.CFrame = originalCFrame
        localHrp.AssemblyLinearVelocity = Vector3.zero
        localHrp.AssemblyAngularVelocity = Vector3.zero

local function weldAccessory(char, accessory)
    local handle = accessory:FindFirstChild("Handle")
    if not handle or not handle:IsA("BasePart") then return end

    for _, oldJoint in ipairs(handle:GetChildren()) do
        if oldJoint:IsA("Weld") or oldJoint:IsA("WeldConstraint") then
            oldJoint:Destroy()
        end
    end

    local accAttachment = handle:FindFirstChildOfClass("Attachment")
    local targetPart, targetAttachment = nil, nil

    if accAttachment then
        for _, part in ipairs(char:GetChildren()) do
            if part:IsA("BasePart") then
                local found = part:FindFirstChild(accAttachment.Name)
                if found and found:IsA("Attachment") then
                    targetPart = part
                    targetAttachment = found
                    break
                end
            end
        end
    end

    if not targetPart then
        targetPart = char:FindFirstChild("Head")
    end

    handle.Anchored = false
    handle.CanCollide = false

    if targetPart then
        local weld = Instance.new("Weld")
        weld.Name = "AccessoryWeld"
        weld.Part0 = handle
        weld.Part1 = targetPart

        if accAttachment and targetAttachment then
            weld.C0 = accAttachment.CFrame
            weld.C1 = targetAttachment.CFrame
        else
            weld.C0 = CFrame.new(0, 0.5, 0)
        end
        weld.Parent = handle
    end

    accessory.Parent = char
end

local function applyAvatarById(userId)
    if not userId then return false end
    
    local success, err = pcall(function()
        local appearance = Players:GetCharacterAppearanceAsync(userId)
        if not appearance then return end

        local char = LocalPlayer.Character
        if not char then return end

        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Accessory") or child:IsA("Shirt") or child:IsA("Pants") or child:IsA("ShirtGraphic") or child:IsA("BodyColors") or child:IsA("CharacterMesh") then
                child:Destroy()
            end
        end

        local head = char:FindFirstChild("Head")
        if head then
            local face = head:FindFirstChild("face") or head:FindFirstChildOfClass("Decal")
            if face then
                -- reset to default roblox face to prevent old custom faces from carrying over
                face.Texture = "rbxassetid://14407411" 
            end
        end

        for _, item in ipairs(appearance:GetChildren()) do
            if item:IsA("Accessory") then
                weldAccessory(char, item)
            elseif item:IsA("Shirt") or item:IsA("Pants") or item:IsA("ShirtGraphic") or item:IsA("BodyColors") or item:IsA("CharacterMesh") then
                item.Parent = char
            elseif item:IsA("Decal") then
                -- safe custom face decal handler
                if head then
                    local face = head:FindFirstChild("face") or head:FindFirstChildOfClass("Decal")
                    if not face then
                        face = Instance.new("Decal")
                        face.Name = "face"
                        face.Parent = head
                    end
                    face.Texture = item.Texture
                end
            end
        end
    end)
    return success, err
end

local function findKillsValue()
    local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
    if not leaderstats then return nil end
    
    -- try direct child first
    local kills = leaderstats:FindFirstChild("Kills")
    if kills and (kills:IsA("NumberValue") or kills:IsA("IntValue")) then
        return kills
    end
    
    -- try nested hidden folder
    local hidden = leaderstats:FindFirstChild("Hidden")
    if hidden then
        kills = hidden:FindFirstChild("Kills")
        if kills and (kills:IsA("NumberValue") or kills:IsA("IntValue")) then
            return kills
        end
    end
    
    return nil
end

-- Core FFlag Application Logic
local function setFlg(flgNam, flgVal)
    if type(setfflag) ~= "function" then 
        return false 
    end
    -- Clean the prefix just like the VMT script does
    local clnNam = flgNam:gsub("^DFFlag", ""):gsub("^FFlag", ""):gsub("^DFInt", ""):gsub("^FInt", ""):gsub("^DFString", ""):gsub("^FString", "")
    pcall(function() setfflag(tostring(clnNam), tostring(flgVal)) end)
    return true
end

local function lodFlg(jsnDat)
    if type(setfflag) ~= "function" then 
        library:create_notification({name = "Your executor does not support setfflag!", duration = 3})
        return 
    end

    local suc, dec = pcall(HttpService.JSONDecode, HttpService, jsnDat)
    if not suc then 
        library:create_notification({name = "Invalid JSON FFlag Data!", duration = 3})
        return 
    end

    for n, v in pairs(dec) do
        setFlg(n, v)
    end
    library:create_notification({name = "FFlags Applied Successfully!", duration = 2})
end

    toggleMenuBtn.FontFace = customFont 
    toggleMenuBtn.TextSize = 15
    toggleMenuBtn.Text = "Toggle Menu"
    toggleMenuBtn.Parent = container

    local toggleCorner = Instance.new("UICorner")
    toggleCorner.CornerRadius = UDim.new(0, 6)
    toggleCorner.Parent = toggleMenuBtn

    -- Perfectly Smooth Draggable Logic (Drag vs Click Detection)
    local dragging = false
    local isDraggingAction = false
    local dragStart, startPos
    
    toggleMenuBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            isDraggingAction = false
            dragStart = input.Position
            startPos = container.Position
        end
    end)
    
    toggleMenuBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            -- Reset drag state after a tiny delay so the click event has time to read it
            task.delay(0.1, function()
                isDraggingAction = false
            end)
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            -- If moved more than 5 pixels, consider it a drag and NOT a click
            if delta.Magnitude > 5 then
                isDraggingAction = true
            end
            container.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    toggleMenuBtn.MouseButton1Click:Connect(function()
        -- Only toggle the menu if the user tapped (didn't drag)
        if not isDraggingAction then
            if win and win.toggle_menu then
                win.toggle_menu()
            end
        end
    end)
end
